  const text=document.getElementById('f-text').value.trim();if(!text)return;

  const data={text,priority:document.getElementById('f-pri').value,member:document.getElementById('f-member').value,date:document.getElementById('f-date').value||null,note:document.getElementById('f-note').value};

  setSyncStatus('syncing');

  try{

    if(editId){await sbUpdate('todos',editId,data);const t=todos.find(x=>x.id===editId);if(t)Object.assign(t,data);}

    else{const rec=await sbInsert('todos',[{id:uid(),workspace:workspaceKey,done:false,created_by:currentUser,...data}]);if(rec&&rec[0])todos.unshift(rec[0]);}

    setSyncStatus('ok');

  }catch(e){setSyncStatus('err');}

  document.getElementById('todo-add-form').style.display='none';editId=null;renderTodos();renderCal();

}


function renderTodos(){

  if(todoFilter==='active')list=todos.filter(t=>!t.done);

  else if(todoFilter==='done')list=todos.filter(t=>t.done);

  else if(todoFilter==='high')list=todos.filter(t=>t.priority==='high');

  else if(todoFilter==='mid')list=todos.filter(t=>t.priority==='mid');

  else if(todoFilter==='low')list=todos.filter(t=>t.priority==='low');

  const el=document.getElementById('todo-list');if(!el)return;

  if(!list.length){el.innerHTML='<div class="empty"><i class="ti ti-inbox" style="font-size:32px;display:block;margin-bottom:8px"></i>沒有符合的任務</div>';updateStats();return;}

  el.innerHTML=list.map(t=>`

    <div class="todo-item">

      <div class="todo-check ${t.done?'done':''}" onclick="toggleTodo('${t.id}')">

        ${t.done?'<i class="ti ti-check" style="font-size:13px"></i>':''}

      </div>

      <div style="flex:1;min-width:0">

        <div class="todo-text ${t.done?'done':''}" style="margin-bottom:4px">${esc(t.text)}</div>

        <div style="display:flex;gap:4px;flex-wrap:wrap;align-items:center">

          ${t.priority?`<span class="badge badge-${t.priority}">${t.priority==='high'?'高優先':t.priority==='mid'?'中優先':'低優先'}</span>`:''}

          ${t.member?`<span class="badge badge-member"><i class="ti ti-user" style="font-size:10px"></i> ${esc(t.member)}</span>`:''}

          ${t.date?`<span class="badge badge-member"><i class="ti ti-calendar" style="font-size:10px"></i> ${t.date}</span>`:''}

          ${t.created_by?`<span style="font-size:11px;color:#aaa">by ${esc(t.created_by)}</span>`:''}

        </div>

      </div>

      <div style="display:flex;gap:4px;flex-shrink:0">

        <button class="btn btn-sm" onclick="showTodoForm('${t.id}')"><i class="ti ti-edit" style="font-size:13px"></i> 編輯</button>

        <button class="btn btn-sm btn-danger" data-del="${t.id}" data-confirm="0" onclick="deleteTodo('${t.id}')"><i class="ti ti-trash" style="font-size:13px"></i></button>

      </div>

    </div>`).join('');

}


async function toggleTodo(id){const t=todos.find(x=>x.id===id);if(!t)return;t.done=!t.done;renderTodos();try{await sbUpdate('todos',id,{done:t.done});setSyncStatus('ok');}catch(e){t.done=!t.done;renderTodos();setSyncStatus('err');}}


async function deleteTodo(id){

  const el=document.querySelector(`[data-del="${id}"]`);if(!el)return;

  if(el.dataset.confirm==='1'){todos=todos.filter(x=>x.id!==id);renderTodos();renderCal();try{await sbDelete('todos',id);setSyncStatus('ok');}catch(e){setSyncStatus('err');loadAll();}}

}


function updateStats(){

  const active=todos.filter(t=>!t.done).length,done=todos.filter(t=>t.done).length,td=todos.filter(t=>t.date===todayStr()&&!t.done).length;

  ['s-active','h-todo'].forEach(id=>{const e=document.getElementById(id);if(e)e.textContent=active;});

  const sd=document.getElementById('s-done');if(sd)sd.textContent=done;

  const hd=document.getElementById('h-today');if(hd)hd.textContent=td;

}

async function saveGoal(){

  const name=document.getElementById('f-gname').value.trim();if(!name)return;

  const newSubs=document.getElementById('f-gsub').value.split('\n').map(s=>s.trim()).filter(Boolean);

  setSyncStatus('syncing');

  try{

    if(editGoalId){

      const g=goals.find(x=>x.id===editGoalId);

      const merged=newSubs.map(s=>{const ex=(g.subs||[]).find(o=>o.text===s);return ex||{text:s,done:false};});

      await sbUpdate('goals',editGoalId,{name,date:document.getElementById('f-gdate').value||null,subs:merged});

      if(g){g.name=name;g.date=document.getElementById('f-gdate').value;g.subs=merged;}

    } else {

      const subs=newSubs.map(s=>({text:s,done:false}));

      const rec=await sbInsert('goals',[{id:uid(),workspace:workspaceKey,name,date:document.getElementById('f-gdate').value||null,created_by:currentUser,subs}]);

      if(rec&&rec[0])goals.unshift(rec[0]);

    }

    setSyncStatus('ok');

  }catch(e){setSyncStatus('err');}

  document.getElementById('goal-add-form').style.display='none';editGoalId=null;renderGoals();

}

function renderCal(){

  const y=calDate.getFullYear(),m=calDate.getMonth();

  const ct=document.getElementById('cal-title');if(!ct)return;

  ct.textContent=`${y} 年 ${m+1} 月`;

  document.getElementById('cal-names').innerHTML=['日','一','二','三','四','五','六'].map(n=>`<div class="cal-day-name">${n}</div>`).join('');

  const first=new Date(y,m,1).getDay(),days=new Date(y,m+1,0).getDate(),ts=todayStr();

  let html='';

  for(let i=0;i<first;i++){const d=new Date(y,m,1-first+i);html+=`<div class="cal-day other-month"><div class="cal-num">${d.getDate()}</div></div>`;}

  for(let d=1;d<=days;d++){

    const ds=`${y}-${p2(m+1)}-${p2(d)}`;const dt=todos.filter(t=>t.date===ds);

    html+=`<div class="cal-day${ds===ts?' today':''}" onclick="showCalDay('${ds}')">

      <div class="cal-num">${d}</div>

      ${dt.slice(0,2).map(t=>`<div class="cal-dot ${t.priority||''}">${esc(t.text)}</div>`).join('')}

      ${dt.length>2?`<div class="cal-dot">+${dt.length-2}</div>`:''}

    </div>`;

  }

  document.getElementById('cal-days').innerHTML=html;

  const mc={};todos.filter(t=>t.date&&!t.done).forEach(t=>{const p=t.date.split('-');if(p.length<2)return;const k=`${p[0]}-${p[1]}`;mc[k]=(mc[k]||0)+1;});

  const curKey=`${y}-${p2(m+1)}`;

  document.getElementById('cal-month-summary').innerHTML=Object.entries(mc).sort(([a],[b])=>a.localeCompare(b)).map(([key,count])=>{

    const[ky,km]=key.split('-');const isActive=key===curKey;

    return `<button onclick="calDate=new Date(${ky},${parseInt(km)-1},1);renderCal()" style="padding:3px 10px;border-radius:100px;border:1px solid ${isActive?'#555':'#ddd'};background:${isActive?'#555':'transparent'};color:${isActive?'#fff':'#888'};cursor:pointer;font-size:12px;font-family:inherit">${parseInt(km)}月 <strong>${count}</strong></button>`;

  }).join('');

}
